# Cosenseで[xxx.icon]にぶら下がったメッセージをミュートする

```js
(() => {
  const MUTED = ['xxx']; // ミュートしたいアイコン名（[xxx.icon] の xxx）
  const STYLE_ID = 'mute-sections-style';

  function update() {
    let style = document.getElementById(STYLE_ID);
    if (!style) {
      style = document.createElement('style');
      style.id = STYLE_ID;
      document.head.appendChild(style);
    }
    if (cosense.Layout !== 'page') { style.textContent = ''; return; }

    const rules = [];
    const seen = new Set();
    for (const line of document.querySelectorAll('#editor .lines > .line')) {
      if (line.classList.contains('line-title')) continue;
      const sec = [...line.classList].find(c => /^section-\d+$/.test(c));
      if (!sec || seen.has(sec)) continue;
      seen.add(sec); // そのセクションの最初の行だけ判定する

      const iconOnly = line.querySelector('.text')?.textContent.trim() === '';
      const alts = [...line.querySelectorAll('img.icon')].map(i => i.alt);
      if (iconOnly && alts.some(a => MUTED.includes(a))) {
        // アイコン行だけ残して、同じセクションの残りを隠す
        rules.push(`#editor .line.${sec}:not(.line-title):not(#${line.id}) { display: none; }`);
        rules.push(`#${line.id} .text::after { content: '（ミュート中）'; color: #999; font-size: .8em; }`);
      }
    }
    style.textContent = rules.join('\n');
  }

  const schedule = () => requestAnimationFrame(update);
  cosense.on('page:changed', schedule);
  cosense.on('lines:changed', schedule);
  schedule();
})();
```
